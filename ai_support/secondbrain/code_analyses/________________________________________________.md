# Версионирование атласа и миграция старых ссылок.

import { useEffect } from 'react';

interface Atlas {
  id: number;
  url: string;
}

const atlas: Atlas[] = [
  { id: 1, url: '/images/old-1.png' },
  { id: 2, url: '/images/old-2.png' },
];

const newAtlas: Atlas[] = [
  { id: 1, url: '/images/new-1.png' },
  { id: 2, url: '/images/new-2.png' },
];

const migrateAtlasReferences = (oldUrls: string[], newUrls: string[]): void => {
  oldUrls.forEach((oldUrl, index) => {
    const atlasEntry = atlas.find(entry => entry.url === oldUrl);
    if (atlasEntry) {
      atlasEntry.url = newUrls[index];
    }
  });
};

const useAtlasMigration = (): void => {
  useEffect(() => {
    migrateAtlasReferences(oldAtlas.map(atlas => atlas.url), newAtlas.map(atlas => atlas.url));
  }, []);
};

export default useAtlasMigration;

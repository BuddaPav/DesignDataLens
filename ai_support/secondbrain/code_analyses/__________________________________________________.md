import { useEffect } from 'react';

const useBrowserSupport = () => {
  useEffect(() => {
    const checkBrowserCompatibility = () => {
      if (
        navigator.userAgent.includes('Chrome') &&
        parseInt(navigator.userAgent.match(/Chrome\/(\d+)/)[1], 10) >= 94
      ) {
        return true;
      } else if (
        navigator.userAgent.includes('Firefox') &&
        parseInt(navigator.userAgent.match(/Firefox\/(\d+)/)[1], 10) >= 86
      ) {
        return true;
      } else if (
        navigator.userAgent.includes('Safari') &&
        parseInt(navigator.userAgent.match(/Version\/(\d+)/)[1], 10) >= 15
      ) {
        return true;
      } else if (
        navigator.userAgent.includes('Edge') &&
        parseInt(navigator.userAgent.match(/Edg\/(\d+)/)[1], 10) >= 94
      ) {
        return true;
      }

      alert(
        'Ваш браузер не поддерживается. Пожалуйста, обновите до последней версии Chrome, Firefox, Safari или Edge.'
      );
      window.location.href = 'https://www.google.com/chrome/';
    };

    checkBrowserCompatibility();
  }, []);
};

export default useBrowserSupport;

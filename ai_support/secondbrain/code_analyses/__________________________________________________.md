# Ярлык «продолжить последнее сохранение» при старте приложения.

```typescript
```typescript
import React, { useEffect } from 'react';

const ContinueLastSaveLink: React.FC = () => {
  useEffect(() => {
    const handleKeyDown = (event: KeyboardEvent) => {
      if (event.key === 'Enter') {
        // Ваш код для продолжения последнего сохранения
        console.log('Продолжить последнее сохранение');
      }
    };

    document.addEventListener('keydown', handleKeyDown);

    return () => {
      document.removeEventListener('keydown', handleKeyDown);
    };
  }, []);

  return (
    <div>
      Нажмите Enter для продолжения последнего сохранения
    </div>
  );
};

export default ContinueLastSaveLink;

type RequestData = {
  // Тип данных для входных параметров запроса

interface ModelResponse {
  // Тип данных для ответа модели
}

class LocalModelClient {
  private model: any; // Предположим, что у нас есть локальная модель

  constructor(model: any) {
    this.model = model;
  }

  async sendRequest(requestData: RequestData): Promise<ModelResponse> {
    const maxTokens = 4096; // Максимальное количество токенов
    const timeout = 10000; // Жёсткий таймаут в миллисекундах

    try {
      const start = Date.now();
      const response: ModelResponse = await new Promise((resolve, reject) => {
        const result = this.model.generate(requestData);
        if (result && result.output) {
          resolve(result.output as ModelResponse);
        } else {
          reject(new Error('No response from model'));
        }
      });

      const duration = Date.now() - start;
      if (duration > timeout) {
        throw new Error(`Request timed out after ${timeout} ms`);
      }

      return response;
    } catch (error) {
      if (error.message === 'No response from model') {
        throw error; // Переиспользуем сообщение ошибки
      }
      throw new Error('Error sending request to local model');
    }
  }
}
